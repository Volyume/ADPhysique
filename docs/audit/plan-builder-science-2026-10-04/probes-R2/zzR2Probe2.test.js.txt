function session(id, endMs, spec, extra = {}) {
  const sets = [];
  for (const [exerciseId, n] of spec) for (let i = 0; i < n; i += 1) sets.push({ exerciseId, setType: 'straight', weight: 50, actualReps: 8 });
  return {
    id, startedAt: endMs - H, endedAt: endMs, durationMinutes: 60, sets,
    weekRirTarget: null, isFirstWeek: false, ratings: { fatigue: null, joint: null, sorenessNext: null }, ...extra,
  };
}

test('probe2: recovery curve, compounding, percent relative to peak', () => {
  const res = {};
  // (A) one chest session, 6 sets, ends at t0
  const t0 = MON9;
  const A = [];
  for (const h of [0, 6, 12, 24, 36, 48, 53, 54, 60, 72]) {
    const m = mapAt([session('a', t0, [['bench', 6]])], t0 + h * H).chest;
    A.push({ hoursAfterEnd: h, pct: m.recoveredPercent, status: m.status, readyInH: m.readyAtMs ? r1((m.readyAtMs - (t0 + h * H)) / H) : null });
  }
  res.A_single_chest_6sets_T60 = A;
  // ready-by offset for the same session read at its end
  const m0 = mapAt([session('a', t0, [['bench', 6]])], t0).chest;
  res.A_readyAfterEndH = r1((m0.readyAtMs - t0) / H);
  res.A_hoursT = m0.contributingSessions[0].hoursT;

  // (B) compounding: Monday 6 sets, then Wednesday 6 sets (48 h later), chest only
  const s1 = session('b1', t0, [['bench', 6]]);
  const s2 = session('b2', t0 + 48 * H, [['bench', 6]]);
  const B = [];
  for (const h of [0, 6, 12, 24, 36, 48, 60, 72, 84]) {
    const at = t0 + 48 * H + h * H;
    const m = mapAt([s1, s2], at).chest;
    B.push({ hoursAfterSecondEnd: h, pct: m.recoveredPercent, status: m.status, readyInH: m.readyAtMs ? r1((m.readyAtMs - at) / H) : null });
  }
  res.B_compounded_chest_two_6set_sessions_48h_apart = B;
  const mB = mapAt([s1, s2], t0 + 48 * H).chest;
  res.B_hoursT = mB.contributingSessions.map((c) => c.hoursT);

  // (C) the same single 6-set vs 12-set session: ready-by after the end
  const c6 = mapAt([session('c6', t0, [['bench', 6]])], t0).chest;
  const c12 = mapAt([session('c12', t0, [['bench', 12]])], t0).chest;
  const c3 = mapAt([session('c3', t0, [['bench', 3]])], t0).chest;
  res.C_readyAfterEndH = { sets3: r1((c3.readyAtMs - t0) / H), sets6: r1((c6.readyAtMs - t0) / H), sets12: r1((c12.readyAtMs - t0) / H) };
  // percent at 24 h for each dose
  res.C_pctAt24h = {
    sets3: mapAt([session('c3', t0, [['bench', 3]])], t0 + 24 * H).chest.recoveredPercent,
    sets6: mapAt([session('c6', t0, [['bench', 6]])], t0 + 24 * H).chest.recoveredPercent,
    sets12: mapAt([session('c12', t0, [['bench', 12]])], t0 + 24 * H).chest.recoveredPercent,
  };

  // (D) biceps vs back after the same single session (the founder's example)
  const sessD = session('d', t0, [['curl', 6], ['row', 6]]);
  const mD = mapAt([sessD], t0 + 24 * H);
  res.D_biceps_vs_back_6setsEach_at24h = {
    biceps: { sets: mD.biceps.lastSessionSets, pct: mD.biceps.recoveredPercent, readyInH: r1((mD.biceps.readyAtMs - (t0 + 24 * H)) / H), hoursT: mD.biceps.contributingSessions[0].hoursT },
    back: { sets: mD.back.lastSessionSets, pct: mD.back.recoveredPercent, readyInH: r1((mD.back.readyAtMs - (t0 + 24 * H)) / H), hoursT: mD.back.contributingSessions[0].hoursT },
  };
  // 'ready after end' for each from the end
  const mD0 = mapAt([sessD], t0);
  res.D_readyAfterEndH = { biceps: r1((mD0.biceps.readyAtMs - t0) / H), back: r1((mD0.back.readyAtMs - t0) / H) };

  // (E) absolute residual at 'ready', in set-equivalents: single vs compounded
  res.E_note = 'ready = residual <= 0.105 x peak; peak(single 6 sets) = 1.0 F = 6 sets; peak(compounded) = 1.2 F';

  fs.writeFileSync(`${OUT}/p2-curve.json`, JSON.stringify(res, null, 1));
  expect(res.A_single_chest_6sets_T60.length).toBe(10);
});

test('probe2b: next-session sentences on the Recovery screen, Home line, projected readiness', () => {
  const out = {};
  const names = { UA: 'Upper A', LA: 'Lower A', UB: 'Upper B', LB: 'Lower B' };
  const planned = {
    UA: { chest: 6, back: 8, side_delts: 4, biceps: 3, triceps: 3 },
    LA: { quads: 8, hamstrings: 6, calves: 4, abs: 3, glutes: 3 },
    UB: { chest: 6, back: 8, rear_delts: 4, biceps: 3, triceps: 3 },
    LB: { quads: 6, hamstrings: 6, glutes: 4, calves: 4, abs: 3 },
  };
  const spec = {
    UA: [['bench', 6], ['row', 4], ['pulldown', 4], ['lateral', 4], ['curl', 3], ['pushdown', 3]],
    LA: [['squat', 8], ['rdl', 6], ['calf', 4], ['crunch', 3]],
    UB: [['bench', 6], ['row', 4], ['pulldown', 4], ['curl', 3], ['pushdown', 3]],
    LB: [['squat', 6], ['rdl', 6], ['calf', 4], ['crunch', 3]],
  };
  const mk = (stateByRoutine) => ['UA', 'LA', 'UB', 'LB'].map((rid, i) => ({
    routineId: rid, name: names[rid], order: i + 1, state: stateByRoutine[rid] ?? SESSION_STATE.OUTSTANDING,
  }));
  const run = (label, sessionsDone, nowMs, stateByRoutine, habit = null) => {
    const map = mapAt(sessionsDone, nowMs);
    const projectedAtMs = nextLikelyTrainingTime({ nowMs, habitualWeekdays: habit?.days ?? null, typicalStartMinute: habit?.minute ?? 18 * 60 });
    const sessions = mk(stateByRoutine);
    const rec = recommendNextWorkout({
      sessions, plannedSetsByRoutine: planned, recoveryMap: map, projectedAtMs, nowMs, routineNamesById: names,
    });
    const wrapped = { ...rec, programmeNextName: names[rec.programmeNext?.routineId] ?? '', routineNamesById: names };
    out[label] = {
      now: stamp(nowMs),
      projectedAt: stamp(projectedAtMs),
      programmeNext: rec.programmeNext?.routineId ?? null,
      recommended: rec.recommended?.routineId ?? null,
      recoveryScreen_answerLine: recoveryAnswerLine(recoveryCounts(map)),
      recoveryScreen_nextSentence: buildNextWorkoutSentence(rec.programmeNext ? wrapped : null, nowMs),
      recoveryScreen_stillToDo: buildStillToDoRows(wrapped, nowMs).map((r) => r.text),
      home_heroLine_programmeNext: rec.programmeNextLine,
      home_heroLine_whenRecommendedPrimary: rec.reason,
      progressRow: buildRecoveryPillarCopy({ map, nowMs, degraded: false }),
      perSession: rec.perSession.map((p) => ({
        routine: p.routineId,
        nowVerdict: p.readinessNow?.verdict, nowMinPct: p.readinessNow?.minPercent, nowLimiting: p.readinessNow?.limitingMuscle,
        nowMuscles: p.readinessNow?.muscles?.map((m) => `${m.muscle}:${m.recoveredPercent}%:${m.status}`),
        projVerdict: p.readinessAtProjected?.verdict, projMinPct: p.readinessAtProjected?.minPercent, projLimiting: p.readinessAtProjected?.limitingMuscle,
        projMuscles: p.readinessAtProjected?.muscles?.map((m) => `${m.muscle}:${m.recoveredPercent}%:${m.status}`),
        line: p.line,
      })),
    };
    return { map, rec };
  };

  // E1: followed in order; Tuesday 09:00 after Upper A on Monday 18:00-19:00
  const tue9 = MON9 + D;
  const uaMon = session('ua', MON9 + 10 * H, spec.UA); // ends Mon 19:00
  run('E1_followedInOrder_TueMorning_afterUpperA', [uaMon], tue9, { UA: SESSION_STATE.COMPLETED });

  // E2: Thursday 09:00: Upper A Mon, Lower A Tue 18:00-19:00; next = Upper B
  const thu9 = MON9 + 3 * D;
  const laTue = session('la', MON9 + D + 10 * H, spec.LA);
  run('E2_followedInOrder_ThuMorning_afterUA_LA', [uaMon, laTue], thu9, { UA: SESSION_STATE.COMPLETED, LA: SESSION_STATE.COMPLETED });

  // E2b: Wednesday 09:00: Upper A Mon, Lower A Tue; next = Upper B, 38 h after UA ended
  const wed9 = MON9 + 2 * D;
  run('E2b_followedInOrder_WedMorning_afterUA_LA', [uaMon, laTue], wed9, { UA: SESSION_STATE.COMPLETED, LA: SESSION_STATE.COMPLETED });

  // E3: out of order: Upper A Monday, Lower B Wednesday 18:00-19:00; Thursday 09:00
  const lbWed = session('lb', MON9 + 2 * D + 10 * H, spec.LB);

  run('E4_weekComplete', [uaMon, laTue], thu9, {
    UA: SESSION_STATE.COMPLETED, LA: SESSION_STATE.COMPLETED, UB: SESSION_STATE.COMPLETED, LB: SESSION_STATE.COMPLETED,

  run('E6_habitMonWedFri_WedMorning_UB_next', [uaMon, laTue], wed9, { UA: SESSION_STATE.COMPLETED, LA: SESSION_STATE.COMPLETED }, { days: [1, 3, 5], minute: 18 * 60 });

  fs.writeFileSync(`${OUT}/p2b-next-session.json`, JSON.stringify(out, null, 1));
  expect(Object.keys(out).length).toBe(7);

test('probe2c: limiting muscle by percent vs the muscle that is ready last', () => {
  const nowMs = MON9;
  // calves trained 4 h ago (6 sets, T 36 h); quads trained 30 h ago (12 sets, T ~102 h)
  const calves = session('calves', nowMs - 4 * H, [['calf', 6]]);
  const quads = session('quads', nowMs - 30 * H, [['squat', 12]]);
  const map = mapAt([quads, calves], nowMs);
  const planned = { quads: 6, calves: 4 };
  const r = sessionReadiness(planned, map);
  const detail = {};
  for (const m of ['quads', 'calves', 'glutes', 'adductors']) {
    detail[m] = map[m] ? { pct: map[m].recoveredPercent, status: map[m].status, readyInH: map[m].readyAtMs ? r1((map[m].readyAtMs - nowMs) / H) : null } : null;
  }
  const res = {
    verdict: r.verdict, minPercent: r.minPercent, limitingMuscle: r.limitingMuscle,
    limitingReadyInH: r1((r.limitingReadyAtMs - nowMs) / H),
    wordsSessionWouldShow: `${r.limitingMuscle}: ${readyClause(r.limitingReadyAtMs, nowMs)}`,
    lastMuscleReadyInH: r1((Math.max(...r.muscles.map((m) => map[m.muscle].readyAtMs)) - nowMs) / H),
    lastMuscleReadyName: r.muscles.map((m) => m.muscle).sort((a, b) => map[b].readyAtMs - map[a].readyAtMs)[0],
    detail,
  };
  fs.writeFileSync(`${OUT}/p2c-limiting.json`, JSON.stringify(res, null, 1));
  expect(res.limitingMuscle).toBeTruthy();
});
