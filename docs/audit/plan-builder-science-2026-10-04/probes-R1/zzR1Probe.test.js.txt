// Probe 2: how a plan's static sets become per-week served sets, and how an
// exercise reaches 6+ sets. Pure replicas of: database.generateInitialPlannedVolume
// (template ramp), sessionAdjustments.getSessionWeeklyAllocation -> coachApply
// computeWeeklySessionAllocation, CoachOutputScreen.handleApplyTraining -> computeVolumeApply.
// ---------------------------------------------------------------------------
import { VOLUME_LANDMARKS } from '../algorithms';
import { computeWeeklySessionAllocation, computeVolumeApply } from '../coachApply';

function templateWeeks() {
  // 6 weeks: 5 accumulation + deload (week 6)  (database.js:5361 rir_ladder [3,2,1,0,0,4], planned_weeks 6)
  const weeks = [];
  for (let w = 1; w <= 6; w += 1) weeks.push({ index: w, rows: {} });
  for (const [muscle, lm] of Object.entries(VOLUME_LANDMARKS)) {
    const acc = 5;
    for (let i = 0; i < acc; i += 1) {
      const progress = acc <= 1 ? 1 : i / (acc - 1);
      weeks[i].rows[muscle] = { planned: Math.round(lm.mev + (lm.mav - lm.mev) * progress), mev: lm.mev, mav: lm.mav, mrv: lm.mrv };
    }
    weeks[5].rows[muscle] = { planned: lm.mev, mev: lm.mev, mav: lm.mav, mrv: lm.mrv };
  }
  return weeks;
}

test('probe: weekly serve of a static plan (template ramp), coach +3, session +1', () => {
  const out = {};
  const weeks = templateWeeks();
  for (const days of [3, 4, 5]) {
    const plan = run({ daysPerWeek: days });
    const perWeek = [];
    for (let wi = 0; wi < 6; wi += 1) {
      const weekMap = Object.fromEntries(Object.entries(weeks[wi].rows).map(([m, r]) => [m, r.planned]));
      const baseMap = Object.fromEntries(Object.entries(weeks[0].rows).map(([m, r]) => [m, r.planned]));
      const sessions = plan.workouts.map((w) => {
        const todays = w.exercises.map((e) => ({
          exerciseId: `${w.name}|${e.exerciseName}`,
          primaryMuscle: byName.get(e.exerciseName)?.primaryMuscle,
          recommendedSets: e.sets,
        }));
        const alloc = computeWeeklySessionAllocation(todays, weekMap, baseMap);
        return { name: w.name, ex: w.exercises.map((e) => ({ n: e.exerciseName, m: byName.get(e.exerciseName)?.primaryMuscle, base: e.sets, served: alloc[`${w.name}|${e.exerciseName}`] })) };
      });
      perWeek.push({ week: wi + 1, plannedChest: weekMap.chest, plannedBack: weekMap.back, sessions });
    }
    out[`d${days}`] = perWeek;
  }
  fs.writeFileSync(`${OUT}/probe2-weekly-serve.json`, JSON.stringify(out, null, 1));
  expect(Object.keys(out).length).toBe(3);
});

// Probe 3: replica of selectExercisesForMuscle's candidate ranking for the
// library path (planEngine.js:1484-1700), to show every score component.
// ---------------------------------------------------------------------------
import { generatePoolFromLibrary } from '../poolGenerator';
import { movementFamily, CLASSIFIED_MUSCLES, familySatisfiesRole } from '../exercise/movementFamily';
import { tierRank, TIER_RANK, AUTO_TIER, isAutoEligible } from '../exercise/canonicality';
import { divisionPoolRule, divisionRoleSpecs } from '../division/profile';
import { SUBREGION_REQUIREMENTS } from '../planEngine';

function normalise(pool) {
  const outp = {};
  for (const [muscle, entries] of Object.entries(pool)) {
    if (!CLASSIFIED_MUSCLES.includes(muscle)) { outp[muscle] = entries; continue; }
    outp[muscle] = entries.map((e) => { const f = movementFamily(e.n, muscle, e.sub); return f === e.sub ? e : { ...e, sub: f }; });
  }
  return outp;
}

function rankReplica(muscle, sessionTarget, weeklyTotal, { goal = 'general', equipment = 'full_gym', experience = 'intermediate', usedNames = new Set(), noIdx = false } = {}) {
  const pool = normalise(generatePoolFromLibrary(rows))[muscle] ?? [];
  let available = pool.filter((e) => e.eq.includes(equipment));
  const rule = divisionPoolRule(goal, muscle);
  if (rule) {
    const restricted = available.filter((e) => {
      if (rule.allowSubs && !rule.allowSubs.includes(e.sub)) return false;
      if (rule.denySubs && rule.denySubs.includes(e.sub)) return false;
      if (rule.denyParams && rule.denyParams.includes(e.p)) return false;
      return true;
    });
    if (restricted.length >= 1) available = restricted;
  }
  available = available.filter((e) => isAutoEligible(e.n));
  const numExHint = Math.max(1, Math.ceil(sessionTarget / 4));
  const recognisable = available.filter((e) => tierRank(e.n) <= TIER_RANK[AUTO_TIER.COMMON]);
  const stage = { afterEquip: pool.filter((e) => e.eq.includes(equipment)).length, afterAuto: available.length, recognisable: recognisable.length };
  if (recognisable.length >= Math.max(2, numExHint)) available = recognisable;
  if (experience !== 'beginner') {
    const gated = available.filter((e) => !/\bassisted\b/i.test(e.n));
    if (gated.length >= Math.max(2, numExHint)) available = gated;
  }
  const req = SUBREGION_REQUIREMENTS[muscle];
  let requiredSubs = [];
  if (req && weeklyTotal >= req.minSets) requiredSubs = [...req.required];
  const roleSpecs = divisionRoleSpecs(goal, muscle);
  const preferredRoles = roleSpecs.map((s) => s.role);
  const satisfies = (e) => requiredSubs.some((r) => familySatisfiesRole(muscle, r, e.sub));
  const scored = available.map((e, idx) => {
    const reqBonus = satisfies(e) ? 0 : 100;
    const paramBonus = ({ heavy_compound: 0, mod_compound: 1, machine: 2, isolation: 3 }[e.p] ?? 3) * 10;
    const roleIndex = preferredRoles.findIndex((role) => e.sub === role);
    const divBonus = roleIndex === -1 ? 0 : -5 / (roleIndex + 1);
    const goalBonus = e.sfr != null ? -(e.sfr / 10) : 0;
    const canonBonus = tierRank(e.n) * 2;
    const idxTerm = noIdx ? 0 : idx;
    return { n: e.n, tier: e.n && (tierRank(e.n) === 0 ? 'staple' : tierRank(e.n) === 1 ? 'common' : 'other'), p: e.p, sub: e.sub, sfr: e.sfr, idx, reqBonus, paramBonus, goalBonus, canonBonus, total: reqBonus + paramBonus + divBonus + goalBonus + canonBonus + idxTerm };
  });
  scored.sort((a, b) => a.total - b.total);
  return { stage, requiredSubs, top: scored.slice(0, 8), poolSize: available.length };
}

test('probe: candidate ranking replica (library path, general/intermediate/full_gym)', () => {
  const cases = [
    ['chest', 3, 6], ['chest', 3, 9], ['back', 5, 10], ['quads', 4, 8], ['hamstrings', 3, 6], ['glutes', 3, 6], ['calves', 3, 7],
    ['biceps', 3, 6], ['triceps', 3, 6], ['side_delts', 4, 8], ['rear_delts', 3, 6], ['abs', 3, 6], ['traps', 3, 6],
  ];
  const res = {};
  for (const [m, st, wk] of cases) {
    res[`${m}@${st}/${wk}`] = { withIdx: rankReplica(m, st, wk), noIdx: rankReplica(m, st, wk, { noIdx: true }) };
  }
  fs.writeFileSync(`${OUT}/probe3-ranking.json`, JSON.stringify(res, null, 1));
  expect(Object.keys(res).length).toBe(cases.length);
});

test('probe: the canonicality pin (staples > 0.6) under corpus order vs name order', () => {

  const helpers = require('./campaign16.helpers');
  const { LIBRARY, inputs, planExerciseNames } = helpers;
  const byNameAsc = [...LIBRARY].sort((a, b) => (a.name < b.name ? -1 : a.name > b.name ? 1 : 0));
  const out = {};
  for (const [label, lib] of [['corpusOrder_asInTest', LIBRARY], ['nameAsc_asShipped', byNameAsc]]) {
    out[label] = {};
    for (const over of [{}, { daysPerWeek: 3 }, { daysPerWeek: 5 }, { experience: 'beginner' }]) {
